cmake_minimum_required(VERSION 3.24)
project(TimeTrace VERSION 1.2.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
configure_file(src/version.h.in generated/version.h @ONLY)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  add_compile_options($<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL> $<$<CONFIG:Release>:/Gy>)
  add_link_options($<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()
add_library(sqlite STATIC third_party/sqlite/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0
  SQLITE_OMIT_DEPRECATED SQLITE_OMIT_JSON SQLITE_OMIT_SHARED_CACHE SQLITE_OMIT_UTF16)
target_include_directories(sqlite PUBLIC third_party/sqlite)
add_library(timetrace_core STATIC src/core.cpp src/storage.cpp src/storage_reminders.cpp src/reminders.cpp src/notification_protocol.cpp)
target_include_directories(timetrace_core PUBLIC src)
target_link_libraries(timetrace_core PUBLIC sqlite)
add_executable(TimeTrace WIN32 src/main.cpp src/platform.cpp src/service.cpp src/integration.cpp src/notifications.cpp src/ui.cpp src/ui_statistics.cpp src/ui_reminders.cpp src/ui_prompts.cpp assets/app.rc)
target_link_libraries(TimeTrace PRIVATE timetrace_core d2d1 dwrite windowscodecs dwmapi shell32 shlwapi ole32 propsys wtsapi32 psapi comctl32 runtimeobject windowsapp version)
target_link_options(TimeTrace PRIVATE /MANIFEST:NO)
target_include_directories(TimeTrace PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
enable_testing()
add_executable(timetrace_tests tests/core_tests.cpp)
target_link_libraries(timetrace_tests PRIVATE timetrace_core)
add_test(NAME core_and_storage COMMAND timetrace_tests)
add_executable(storage_benchmark tests/storage_benchmark.cpp)
target_link_libraries(storage_benchmark PRIVATE timetrace_core)
add_executable(integration_tests tests/integration_tests.cpp src/integration.cpp)
target_link_libraries(integration_tests PRIVATE timetrace_core shell32 ole32 propsys advapi32)
target_include_directories(integration_tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_test(NAME installation_registration COMMAND integration_tests)
add_executable(foreground_probe WIN32 tests/foreground_probe.cpp)
target_link_libraries(foreground_probe PRIVATE user32 shell32)
add_executable(resource_probe tests/resource_probe.cpp)
target_link_libraries(resource_probe PRIVATE psapi)
add_executable(lifecycle_probe tests/lifecycle_probe.cpp)
target_link_libraries(lifecycle_probe PRIVATE psapi advapi32 sqlite)
add_executable(notification_probe tests/notification_probe.cpp)
target_link_libraries(notification_probe PRIVATE runtimeobject windowsapp)
add_executable(reminder_actions_probe tests/reminder_actions_probe.cpp)
target_link_libraries(reminder_actions_probe PRIVATE timetrace_core user32)
add_test(NAME notification_actions COMMAND reminder_actions_probe $<TARGET_FILE:TimeTrace> ${CMAKE_BINARY_DIR}/notification-actions.json)
add_executable(service_tests tests/service_tests.cpp src/service.cpp src/platform.cpp src/integration.cpp)
target_include_directories(service_tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(service_tests PRIVATE timetrace_core shell32 ole32 propsys version advapi32)
add_test(NAME service_queue COMMAND service_tests)
